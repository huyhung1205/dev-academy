void selectionSort(int a[], int n){
for(int i = 0; i < n; i++){
int min_pos = i;
for (int j = i + 1; j < n; j++){
if (a[j] < a[min_pos]){
min_pos = j;
}
}
swap(a[min_pos], a[i]);
}
}
Độ phức tạp: O(N^2)

void bubbleSort(int a[], int n){
for(int i = 0; i < n; i++){
for(int j = 0; j < n - i - 1; j++){
if (a[j] > a[j + 1]){
swap(a[j], a[j + 1]);
}
}
}
}
Độ phức tạp: O(N^2)

3.Thuật toán sắp xếp chèn (Insertsion sort):
void insertionSort(int a[], int n){
for(int i = 1; i < n; i++){
int pos = i - 1, x = a[i];
while(pos >= 0 && a[pos] > x){
a[pos + 1] = a[pos];
--pos;
}
a[pos + 1] = x;
}
}
Độ phức tạp: O(N^2)
