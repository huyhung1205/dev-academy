int firstPos(int a[], int n, int x){
int res = -1, left = 0, right = n - 1;
while(left <= right){
int mid = (left + right) / 2;
if(a[mid] == x){
res = mid; // cập nhật
//Tìm thêm đáp án tốt hơn
right = mid - 1;
}
else if(a[mid] < x){
left = mid + 1;
}
else{
right = mid - 1;
}
}
return res; Độ phức tạp: O(logN)
}

int lower(int a[], int n, int x){
int res = -1;
int left = 0, right = n - 1;
while(left <= right){
int mid = (left + right) / 2;
if(a[mid] >= x){
res = mid; // cập nhật

}
else{

}

SỬ DỤNG LOWER_BOUND
Lower_bound là một hàm tương Tương tự như hàm sort, lower_bound cũng trả
tự như mục 5 ở trên, nó có thể áp về iterator chứ không trả về giá trị.
dụng cho mảng, vector, set, map.
Nó trả về vị trí đầu tiên của phần tử lớn hơn
Điều kiện áp dụng: Mảng, vector hoặc bằng X, nếu trong mảng, vector bạn tìm
đã được sắp xếp tăng dần kiếm không có phần tử lớn hơn hoặc bằng X
(tất cả đều nhỏ hơn X) thì lower_bound trả về
