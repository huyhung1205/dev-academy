DFS trên mảng 2 chiều
int n, m;
int a[1005][1005];
bool visited[1005][1005];
int dx[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy[8] = {-1, 0, 1, -1, 1, -1, 0, 1};
void DFS(int i, int j){
visited[i][j] = true;
for(int k = 0; k < 8; k++){
int i1 = i + dx[k], j1 = j + dx[k];
if(i1 >= 1 && i1 <= n && j1 >= 1 && j1 <= m && !visited[i1][j1]){
DFS(i1, j1);
}
}
}
